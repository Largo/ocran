$gem install ocran
terminal
# one self-extracting executable: app.exe on Windows, app on Linux/macOS
$ ocran app.rb
 
# a portable folder, a macOS .app bundle, a Windows installer
$ ocran --output-dir myapp/ app.rb
$ ocran --macosx-bundle --bundle-id com.example.myapp app.rb
$ ocran --icon app.ico --innosetup app.iss app.rb
 
# experimental: one binary for Linux, macOS and Windows, x86-64 and ARM
$ ocran app.rb --cosmo-ruby ./ruby.com
